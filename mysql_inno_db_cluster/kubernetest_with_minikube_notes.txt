Kubernetes with minikube:-

********************* to start minikube cluster with more than one node *******************************

minikube start --nodes 3 -p ecom-multi-node-cluster

--to start minikube with one node cluster (default)
minikube start 

-- command to see details of a node
 kubectl describe node <node_name>
 
-- kubernetes cluster status  
 minikube status -p ecom-multi-node-cluster
 
-- to run kyebernetes yml file
kubectl apply -f your-file-name.yaml

kubectl apply -f mysql_kubernetes.yml
kubectl apply -f mysql_kubernetes.yml

-- to run kuberenetes dashboard 

minikube dashboard -p ecom-multi-node-cluster --url 




kubectl exec -it <your-pod-name> -- bin/bash

kubectl exec -it mysql-0 -- bin/bash
kubectl exec -it mysql-1 -- bin/bash


-- delete kubernetes cluster by the name ecom-multi-node-cluster
minikube delete -p ecom-multi-node-cluster


kubectl drain --force ecom-multi-node-cluster --ignore-daemonsets --delete-emptydir-data
kubectl drain --force ecom-multi-node-cluster-m02 --ignore-daemonsets --delete-emptydir-data
kubectl drain --force ecom-multi-node-cluster-m03 --ignore-daemonsets --delete-emptydir-data


kubectl delete node ecom-multi-node-cluster
kubectl delete node ecom-multi-node-cluster-m02
kubectl delete node ecom-multi-node-cluster-m03

kubectl scale statefulset mysql --replicas=0
